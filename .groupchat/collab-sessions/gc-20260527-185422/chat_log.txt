def _recall_wing_semantic(wing: str, query: str, n_results: int = 5) -> str:
    """Recall drawers from a single wing via semantic search.

    Returns formatted string (empty if nothing above threshold or on error).
    """
    try:
        from mempalace.searcher import search_memories  # noqa: PLC0415

        result = search_memories(query, wing=wing, n_results=n_results)
        if not result or not result.get("results"):
            return ""

        lines: list[str] = []
        for item in result["results"]:
            sim = item.get("similarity", 0)
            if sim < _MIN_SIM:
                continue
            room = item.get("room", "?")
            text = item.get("text", "")
            lines.append(f"  [sim={sim:.2f}|{room}] {text}")

        if not lines:
            return ""

        output = "\n".join(lines)
        if len(output) > _MAX_CHARS_PER_WING:
            output = output[:_MAX_CHARS_PER_WING] + "\n  ... (truncated)"
        return output
    except Exception as e:
        logger.warning("auto_recall: semantic search failed for wing={}, falling back: {}", wing, e)
        return _recall_wing_fallback(wing)

def _recall_wing_fallback(wing: str, n_results: int = 5) -> str:
    """Fallback: use MemoryStack.recall() (no semantic filtering)."""
    try:
        from mempalace.layers import MemoryStack  # noqa: PLC0415

        stack = MemoryStack()
        result = stack.recall(wing=wing, n_results=n_results)

            return ""
        if len(result) > _MAX_CHARS_PER_WING:
            result = result[:_MAX_CHARS_PER_WING] + "\n... (truncated)"
        return result

        logger.warning("auto_recall: fallback recall failed for wing={}: {}", wing, e)
