    engine,
    gc_settings,
    agents,
    leader_name=None,
):
    """Top-level broadcast orchestrator.

    Runs broadcast rounds until discussion completes.
    Returns BroadcastResult with summary_data and discussion_complete flag.
    """
    result = BroadcastResult()
    
    while True:
        summary_data, discussion_complete = await broadcast_round(
            engine, gc_settings, agents, leader_name
        )
        result.summary_data = summary_data
        
        if discussion_complete:
            result.discussion_complete = True
            break
            
        # Brief yield to allow other tasks to run
        await asyncio.sleep(0)
    
    return result
