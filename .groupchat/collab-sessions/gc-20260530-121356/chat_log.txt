const C=document.getElementById('C'),X=C.getContext('2d');
const W=800,H=600;C.width=W;C.height=H;
let score=0,lives=3,level=1,combo=0,gameOver=false,running=false;
const paddleW=120,paddleH=14,ballR=8;
let px=W/2-60,py=H-30,bx=W/2,by=H-50,bdx=4,bdy=-4;
let stars=[],particles=[];
for(let i=0;i<80;i++)stars.push({x:Math.random()*W,y:Math.random()*H,r:Math.random()*2,a:Math.random()});

// Brick layout
const colors=['#f0f','#a0f','#08f','#0ff','#0f0','#ff0','#f80'];
const rows=7,cols=10,bw=65,bh=20,gap=8,topOff=60;
let bricks=[];

function initBricks(){
  bricks=[];
  for(let r=0;r<rows;r++)for(let c=0;c<cols;c++){
    let x=c*(bw+gap)+35,y=r*(bh+gap)+topOff;
    bricks.push({x,y,w:bw,h:bh,color:colors[r%7],alive:true,hp:r<2?2:1});
  }
}

function resetBall(){bx=W/2;by=H-50;bdx=(4+level*0.5)*(Math.random()>.5?1:-1);bdy=-(4+level*0.5)}

function spawnParticles(x,y,color,n=15){
  for(let i=0;i<n;i++)particles.push({x,y,vx:(Math.random()-.5)*6,vy:(Math.random()-.5)*6,life:1,color})
}

function startGame(){
  score=0;lives=3;level=1;combo=0;gameOver=false;running=true;
  document.getElementById('overlay').style.display='none';
  initBricks();resetBall();
}

function update(){
  if(!running||gameOver)return;
  // paddle follow mouse
  if(mouseX>0)px=Math.max(0,Math.min(W-paddleW,mouseX-paddleW/2));
  
  bx+=bdx;by+=bdy;
  if(bx<=ballR||bx>=W-ballR)bdx=-bdx;
  if(by<=ballR)bdy=-bdy;
  if(by>H){lives--;document.getElementById('livesVal').textContent=lives;combo=0;document.getElementById('comboVal').textContent=0;if(lives<=0){gameOver=true;showOverlay()}else resetBall()}
  
  // paddle collision
  if(by+ballR>=py&&by+ballR<=py+paddleH&&bx>px&&bx<px+paddleW){
    let hit=(bx-px)/paddleW;
    let angle=hit*Math.PI*0.66-Math.PI*0.33;
    let spd=Math.sqrt(bdx*bdx+bdy*bdy);
    bdx=spd*Math.sin(angle);bdy=-spd*Math.cos(angle);
    spawnParticles(bx,py,'#0ff',8);
  }
  
  // brick collision
  for(let b of bricks){
    if(!b.alive)continue;
    if(bx>b.x&&bx<b.x+b.w&&by> b.y&&by<b.y+b.h){
      bdy=-bdy;
      if(--b.hp<=0){
        b.alive=false;
        combo++;score+=100*combo;
        spawnParticles(b.x+b.w/2,b.y+b.h/2,b.color,20);
      }else{
        combo=0;
        spawnParticles(b.x+b.w/2,b.y+b.h/2,b.color,8);
      }
      document.getElementById('scoreVal').textContent=score;
      document.getElementById('comboVal').textContent=combo;
      break;
    }
  }
  
  // level clear
  if(bricks.every(b=>!b.alive)){
    level++;
    document.getElementById('levelVal').textContent=level;
    initBricks();resetBall();
  }
  
  // particles
  for(let i=particles.length-1;i>=0;i--){
    let p=particles[i];
    p.x+=p.vx;p.y+=p.vy;p.life-=0.025;
    if(p.life<=0)particles.splice(i,1);
  }
}

function draw(){

  stars.forEach(s=>{X.globalAlpha=Math.abs(s.a)*0.6;X.fillStyle="#fff";X.beginPath();X.arc(s.x,s.y,s.r,0,Math.PI*2);X.fill()});
  X.globalAlpha=1;
  for(let b of bricks){if(!b.alive)continue;
    X.shadowColor=b.color;X.shadowBlur=15;
    X.fillStyle=b.color;X.fillRect(b.x,b.y,b.w,b.h);
    X.shadowBlur=0;
    X.strokeStyle="rgba(255,255,255,0.3)";X.lineWidth=1;X.strokeRect(b.x+0.5,b.y+0.5,b.w-1,b.h-1)}
  ball.trail.forEach(t=>{X.globalAlpha=t.a*0.4;X.fillStyle='#0ff';X.beginPath();X.arc(t.x,t.y,ballR*0.6,0,Math.PI*2);X.fill()});
  X.globalAlpha=1;
  X.shadowColor='#0ff';X.shadowBlur=20;
  X.fillStyle='#0ff';X.beginPath();X.arc(bx,by,ballR,0,Math.PI*2);X.fill();
  X.shadowBlur=0;
  X.shadowColor='#0ff';X.shadowBlur=10;
  let grd=X.createLinearGradient(px,py,px+paddleW,py);
  grd.addColorStop(0,'#0ff');grd.addColorStop(.5,'#f0f');grd.addColorStop(1,'#0ff');
  X.fillStyle=grd;X.shadowBlur=15;X.fillRect(px,py,paddleW,paddleH);
  X.shadowBlur=0;
  particles.forEach(p=>{X.globalAlpha=p.life;X.fillStyle=p.color;X.shadowColor=p.color;X.shadowBlur=10;X.fillRect(p.x-2,p.y-2,4,4)});
  X.globalAlpha=1;X.shadowBlur=0;
}

  let o=document.getElementById('overlay');
  o.style.display='flex';

}
